\begin{abstract}
GO-10 measured the complementarity tax two consumers pay on one static
source; GO-11 solved the exact region behind it, through the $m=2$
frontier system with its $2\times2$ matrix water levels; GO-12 showed
the staleness of the eraser's context is an access-width phenomenon,
reduced entirely to one conditional variance $q_{\mathcal G}$. GO-13
asks the merged question: \emph{what do two consumers pay when the
eraser's context ages?} Nothing here is a claim; the conjectures are
gates for future registrations.
\end{abstract}

\section{The problem}

Take GO-11's $m=2$ setup verbatim (whitened latent $Z\in\mathbb
R^{2+r}$, reads $y_A,y_B$, context block $V$ now a stationary AR(1)
\emph{process} with observations $S_s=V_s+U_s$, $\mathrm{Var}\,U=
\tau^2$), single-letter two-dimensional records
$\hat{\mathbf Y}=AZ+N$ with per-consumer distortions $D_A,D_B$. The
eraser holds a context $\sigma$-algebra $\mathcal G$ from the
observation process (slice, causal prefix, full path --- GO-12's access
family). Coordinates: the joint conditional coordinate
$L_{AB}(\mathcal G)$ (weighted-program value of GO-11 Theorem~9 with
the conditional covariance induced by $\mathcal G$), the single-consumer
coordinates $L_A(\mathcal G),L_B(\mathcal G)$, and the taxes
\[
\mathrm{CT}_R \;=\; R_{AB}-\max(R_A,R_B),\qquad
\mathrm{CT}_W(\mathcal G)\;=\;L_{AB}(\mathcal G)-
\max\bigl(L_A(\mathcal G),L_B(\mathcal G)\bigr).
\]
$\mathrm{CT}_W$ is now a \emph{function of the eraser's access}: the
dynamic complementarity tax.

\section{Conjectures}

\newtheorem{theorem}{Theorem}
\begin{theorem}[The matrix-$q$ reduction; settled and netted,
GO-P-2026-067]\label{thm:matq}
Hypotheses (all load-bearing, per the R-IND-5 pass): joint
Gaussianity; record noise $N\perp(T,\mathcal G)$; the
conditional-independence embedding $\mathrm{Cov}(T,V_s)=
\mathrm{Cov}(T,V_t)\,a^{|t-s|}$; $\mathcal G$ generated by a nonempty
subset of the observations $\{S_s\}$; weight $w\in[0,1]$. Then for
$m=2$ single-letter records, $\mathcal G$ enters all coordinates only
through $\Sigma_{T\mid\mathcal G}=\Sigma_T-(1-q_{\mathcal G})\,
cc^{\mathsf T}$ ($c=\mathrm{Cov}(T,V_t)$, $q_{\mathcal G}=
\mathrm{Var}(V_t\mid\mathcal G)$): the coordinates are GO-11
Theorem~9's program at $(\Sigma_T,\Sigma_{T\mid\mathcal G})$ ---
GO-12's channel-agnostic sufficiency argument transfers verbatim to
2-D records. Consequently, for \emph{scalar} context every coordinate
and both taxes are functions of $q_{\mathcal G}$ alone
(\textbf{access-class universality}: equal-$q$ classes give identical
taxes --- and only classes at \emph{attainable} common $q$ compare,
each class having its own $q_{\min}$); the coordinates are
nondecreasing in $q$ (Loewner: $\partial_q\Sigma_{T\mid\mathcal G}
=cc^{\mathsf T}\succeq0$, pointwise in every channel, feasible set
$q$-free); and at $q\to1$ they converge to the marginal program
(continuity of the min from concavity of the value in $q$), so
$\mathrm{CT}_W\to\mathrm{CT}_R$. Scalar universality \emph{fails} for
vector context $r\ge2$ (verifier counterexample: equal
$\mathrm{tr}\,\mathrm{Cov}(V\mid\mathcal G)$ with coordinates
differing by $0.12$); the reduction survives only to the matrix
$\mathrm{Cov}(V\mid\mathcal G)$. [R-IND-5: PASS, reduction to
$3.6\times10^{-15}$, universality triples to $7\times10^{-12}$;
governed harness 067 ALL PASS 4/4, seed 20260921.]
\end{theorem}

\begin{conjecture}[The tax curve; monotone clause REFUTED, shape open]
\label{conj:taxflow}
$\mathrm{CT}_W(q)$ runs continuously from GO-10's work tax to
$\mathrm{CT}_R$ as $q\to1$ (both endpoints now proven,
Theorem~\ref{thm:matq}). The v0.1 monotone clause is \textbf{refuted}:
the R-IND-5 pass exhibited a regime with the tax monotone
\emph{increasing} by $0.16$ nats ($\rho$'s $=(-0.2,0.5,0.6)$,
$D=(0.3,0.5)$, $w=0.5$) and a non-monotone instance with a kink
exactly where $\arg\max(L_A,L_B)$ switches consumers --- the curve is
continuous but only piecewise smooth. Near-invariance (variation
$<5\times10^{-4}$ bits) holds at one-dominant-consumer instances but
is not generic. The open object is the \emph{characterization}: which
$(\rho_{AB},\rho_{AV},\rho_{BV},D_A,D_B,w)$ regimes make staleness
raise, lower, or leave the tax --- with the max-switch locus as the
organizing boundary.
\end{conjecture}

convolution parameter)?
