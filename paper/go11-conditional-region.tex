\title{GO-11 (problem statement): The Exact Two-Consumer\\
Rate--Work Region with Reset Side Information}
\author{Andrew H. Bond}
\date{Problem statement v0.8 --- August 2026}
\begin{document}\maketitle

\begin{abstract}
The complementarity-tax note proved \emph{floors} for the joint description
of two consumers whose reset may consult side information; those floors are
tight only as $D\to0$. This document poses the exact-region problem. The
coordinate pair itself is not new: per test channel the rate--work gap is
$I(\hX;S)$ (a chain-rule identity that is working folklore in the
rate--distortion--equivocation literature since Yamamoto, and appears inside
Paper V's achievability proof), and exact scalar-Gaussian regions carrying a
conditional-information coordinate with \emph{one} distortion constraint
exist in print --- most sharply G\"unl\"u et al.'s conditional-leakage
region under a degradedness ordering, and Wang et al.'s
rate--distortion--classification function, which is the scalar instance of
our frontier. What our sweeps did not find, and what GO-11 asks for, is the
region when (i) the source is a vector read through \emph{two} simultaneous
weighted-quadratic consumer distortions, (ii) $S$ is arbitrary jointly
Gaussian --- neither aligned with the read modes nor degraded relative to
any decoder observation --- and (iii) the frontier comes with the
semidefinite condition under which the complementarity-tax work floor is
attained. Even the single-consumer vector endpoint (a vector-Gaussian
common-reconstruction function) has no published solution. We state the
conjectured structure, the first decidable step, and the verification
obligations a registration must meet.
\end{abstract}

\section{The problem}
$X\sim\mathcal N(0,\Sigma_x)$ on $\mathbb R^d$, i.i.d.; consumers $A,B$ with
rank-one reads $u,v$ (general PSD $P_1,P_2$ as the extension); reset side
information $S$, jointly Gaussian with $X$, available only at erasure, with
the model Markov chain $(\hX_A,\hX_B)-X-S$ \cite{bondlandauerV}. Define, for
$0<D_A\le\sigma_u^2$, $0<D_B\le\sigma_v^2$,
\[
\mathcal{RW}(D_A,D_B\mid S)=\operatorname{cl}\!\bigcup_{p(\hat x_A,\hat x_B\mid x)}
\bigl\{(R,L):\ R\ge I(X;\hX_A,\hX_B),\ L\ge I(X;\hX_A,\hX_B\mid S)\bigr\},
\]
the union over joint test channels with $\E\,d_A\le D_A$, $\E\,d_B\le D_B$.

\textbf{Problem (GO-11).} Characterize $\mathcal{RW}(D_A,D_B\mid S)$
exactly --- its Pareto frontier, the optimal channel family, and the
condition under which the complementarity-tax floors \cite{bondtaxnote} are
attained.

\textbf{What is known (post-sweep accounting).}
(i) \emph{The one-constraint pair-region is solved, in general.} Paper V's
main region theorem characterizes
$\{(I(X;\hX),I(X;\hX\mid S))\}$ under a single distortion constraint for
general alphabets, and its Pareto-channel equation is a Blahut--Arimoto-type
fixed point for the weighted objective
$\alpha I(X;\hX)+(1-\alpha)I(X;\hX\mid S)+\beta\,\E d$ --- which, by
Fact~\ref{fact:gap}, \emph{is} the scalarization \eqref{eq:dcib} below with
$\beta'=1-\alpha$. Its solved Gaussian instances: the scalar corner (any
jointly Gaussian scalar $S$) and the mode-aligned vector case (independent
modes, per-mode $S$, one aggregated constraint) with its reset-water-filling
frontier \cite{bondlandauerV}. (ii) \emph{The rate-only projection} is the
simultaneous-service max-det program (\cite{bondtwoobserver}; after the
read-plane reduction, the Xiao--Luo / semidefinite-condition literature).
(iii) \emph{The floors}: $R\ge\tfrac12\log^+(\kappa/(D_AD_B))$,
$L\ge\tfrac12\log^+(\kappa_S/(D_AD_B))$, floor difference exactly $I(Y;S)$,
$Y=B^{\mathsf T}X$; rate floor exact iff
$\mathrm{diag}(D_A,D_B)\preceq B^{\mathsf T}\Sigma_xB$; work floor generally
strict at $D>0$ \cite{bondtaxnote}. (iv) \emph{Operationally}, decode
thresholds realize $0.4$--$0.8\times$ of the floor-level gaps at finite $n$
on two source families (GO-P-2026-058/059); an exact region would say what
the remaining fraction is made of. (v) \emph{Scalar exact regions with one
distortion constraint exist in the secrecy literature} --- see
Section~\ref{sec:delineation} --- including the conditional-leakage
coordinate itself under degradedness (G\"unl\"u et al.) and with the
non-degraded scalar Gaussian converse explicitly left open
(Villard--Piantanida). GO-11's open content is therefore exactly: general
(non-aligned, non-degraded) Gaussian $S$ --- already for \emph{one}
consumer --- and everything two-consumer beyond Paper V's structural union.

\section{The organizing identity}
\begin{fact}[chain rule; folklore]\label{fact:gap}
For any test channel with $(\hX_A,\hX_B)-X-S$ Markov, writing
$\hX=(\hX_A,\hX_B)$,
\[
I(X;\hX)-I(X;\hX\mid S)\;=\;I(\hX;S).
\]
\end{fact}
\begin{proof}
$I(\hX;S)=I(\hX;X,S)-I(\hX;X\mid S)=I(\hX;X)+I(\hX;S\mid X)-I(\hX;X\mid S)$,
and $I(\hX;S\mid X)=0$ by the Markov chain.
\end{proof}

\begin{remark}[attribution]
Fact~\ref{fact:gap} is not ours: it is used verbatim inside Paper V's
achievability proof (the binning step) and is working folklore throughout
the rate--distortion--equivocation lineage (Yamamoto; Villard--Piantanida's
region manipulations). What it \emph{organizes} here is the observation
that the region is generated by a single curve of corners
$(R,L)=(I,\,I-I(\hX;S))$, so the frontier is the scalarization
\begin{equation}
\min_{p(\hat x\mid x)}\ I(X;\hX)-\beta'\, I(\hX;S)
\quad\text{s.t.}\quad \E d_A\le D_A,\ \E d_B\le D_B,
\qquad \beta'\in[0,1],
\label{eq:dcib}
\end{equation}
a distortion-constrained information bottleneck with the reset context $S$
as relevance --- equivalently, in equivocation coordinates, a
maximize-the-leakage member of the Yamamoto family with the orientation
that \emph{helps} the $S$-holding party. $\beta'=0$ recovers the
simultaneous-service rate program; $\beta'=1$ the work-only endpoint.
\end{remark}

\begin{remark}[convexity boundary --- a warning]\label{rem:convex}
For $\beta'\in[0,1]$, \eqref{eq:dcib} is the convex combination
$\alpha I(X;\hX)+(1-\alpha)I(X;\hX\mid S)$ of the two coordinates
($\alpha=1-\beta'$), each convex in the channel (the conditional coordinate
by the per-$s$ decomposition verified in VI-8), so the scalarization stays
convex and traces the entire frontier. The difference form
$I(X;\hX)-\beta' I(\hX;S)$ is \emph{not} manifestly convex on its own
($I(\hX;S)$ is convex in the channel, not concave); any argument pushing
$\beta'>1$ leaves the convex regime and must be justified separately.
\end{remark}

\section{Conjectures and questions}
The sweeps (Section~\ref{sec:delineation}) make the following
\emph{expected by analogy} rather than surprising: every ingredient exists
separately in print; no assembled solution does.

\begin{conjecture}[Gaussian sufficiency]\label{conj:gauss}
The frontier of \eqref{eq:dcib} is attained by jointly Gaussian channels
$\hX = KX+W$; consequently the region is the closure of
$\{(R(\Sigma),\,R(\Sigma)-I_{\Sigma}(\hX;S))\}$ over error covariances
$0\preceq\Sigma\preceq\Sigma_x$ meeting the trace constraints, with both
coordinates explicit determinant functionals of $\Sigma$. \emph{Expected
route: extend the conditional-EPI arguments that prove Gaussian optimality
in the degraded scalar cases (G\"unl\"u et al.; Sankar et al.'s
simultaneous rate-and-leakage optimality) and the linear-policy optimality
machinery of the mixed-quadratic Gaussian literature; the open difficulty
is precisely the non-degraded, non-aligned coupling, where
Villard--Piantanida's scalar converse is already open.}
\end{conjecture}

\begin{conjecture}[Two-consumer reset water-filling --- possibly
optimistic]\label{conj:frontier}
In the $S$-whitened read frame, the frontier is traced by an
$\alpha$-parametrized allocation program generalizing Paper V's reset
water-filling: the $\alpha=1$ endpoint re-derives the max-det rate optimum,
the $\alpha=0$ endpoint tilts distortion toward the directions of the read
plane that $S$ resolves least (the operational allocation signature of
GO-P-2026-058/059), and for scalar-pair instances the program reduces to a
two-mode quadratic of the Paper V vector-frontier type. \emph{Caution: with
$\Sigma_x$, $\Sigma_{X\mid S}$, and the read operators not simultaneously
diagonalizable, three mutually non-commuting matrices compete; the
analogous regime of the 2026 rate--distortion--deception work found no
closed form. A clean water-filling may exist only under an alignment
condition, with the general case a genuinely semidefinite frontier.}
\end{conjecture}

\begin{conjecture}[Exactness condition for the work floor]\label{conj:tight}
The work floor $\tfrac12\log(\kappa_S/(D_AD_B))$ is attained iff a
semidefinite condition of the form
$\mathrm{diag}(D_A,D_B)\preceq B^{\mathsf T}\Sigma_{X\mid S}B$ holds
\emph{and} the optimal channel decouples from $S$'s residual --- expected
to fail at every $D>0$ when $S$ has an $X$-independent component
($\tau^2>0$; matching the strictness remark of \cite{bondtaxnote} and the
scalar corner's $D>0$ gap), with equality exactly in the $D\to0$ limit.
The qualifier is forced by Proposition~\ref{prop:marg}: an $X$-measurable
$S$ is maximally informative yet the single-consumer conditional floor is
attained there at every $D\le\tfrac12$ --- the strictness belongs to the
encoder's ignorance of $S$'s residual, not to informativeness.
\end{conjecture}

\begin{question}[Single-consumer vector case; endpoint = a vector-Gaussian
common-reconstruction function]\label{q:single}
Even $m=1$ is open: for one rank-one read of a $d$-dimensional source with
\emph{arbitrary} jointly Gaussian $S$, what is the frontier --- and already
its work-only endpoint $L_A(D\mid S)=\min I(X;\hX\mid S)$, which equals
Steinberg's common-reconstruction rate--distortion function? The targeted
sweep found \emph{no published vector-Gaussian CR function under any
formulation} --- not even unweighted distortion, not even the mode-aligned
case solved in Paper V. This is Paper V's declared ``general Gaussian
coupling'' open item, specialized to rank-one reads.

\emph{The first decidable step (rank-one $P=ww^{\mathsf T}$) is now
settled} --- see Proposition~\ref{prop:marg}: marginalization is strictly
suboptimal, the rank-one case does \emph{not} collapse to the scalar
corner, and the mechanism (the encoder's implicit knowledge of the
$X$-measurable part of $S$) reshapes what the general answer must look
like.
\end{question}

\section{The marginalization dichotomy, settled}\label{sec:marg}

Throughout this section $m=1$: one rank-one read $Y=w^{\mathsf T}X$,
distortion $\E(Y-\hY)^2\le D$. Two reductions first. \emph{(WLOG scalar
reproduction.)} Replacing $\hX$ by $\hY:=w^{\mathsf T}\hX$ leaves the
distortion unchanged and can only decrease $I(X;\cdot\mid S)$ (conditional
data processing), and $\hY-X-S$ persists; so reproductions are scalar.
\emph{(Marginalized class $=$ scalar corner.)} If the channel depends on
$X$ only through $Y$ (i.e., $\hY-Y-(X,S)$), then given $(Y,S)$, $\hY$ is
independent of $X$, so $I(X;\hY\mid S)=I(Y;\hY\mid S)$, and the class
optimum is exactly the scalar corner value of \cite{bondlandauerV} for the
pair $(Y,S)$ --- the corner's converse applies to every $p(\hat y\mid y)$,
discrete or not.

\begin{proposition}[Marginalization is strictly suboptimal]\label{prop:marg}
Let $d=2$, $X\sim\mathcal N(0,I_2)$, $Y=X_1$, and $S=X_1+X_2$ (so
$\rho^2_{YS}=\tfrac12$ and $S$ is $X$-measurable). For $0<D<1$:
\begin{enumerate}
\item[(i)] the marginalized-class optimum is the scalar corner
$L_{\mathrm{marg}}(D)=\tfrac12\log_2\!\frac{1+D}{2D}$;
\item[(ii)] the vector problem has the exact value
\[
L_{\mathrm{vec}}(D)\;=\;\tfrac12\log_2^{+}\!\frac{1}{2D}
\;=\;R_{Y\mid S}(D),
\]
Gray's conditional rate--distortion function of $(Y,S)$
($\operatorname{Var}(Y\mid S)=\tfrac12$), attained by the linear channel
$\hY=(1-D)X_1+DX_2+N$, $N\sim\mathcal N(0,D(1-2D))$, for $D\le\tfrac12$,
and by $\hY=S/2$ (zero conditional content) for $D\ge\tfrac12$;
\item[(iii)] hence the strict gap
$L_{\mathrm{marg}}(D)-L_{\mathrm{vec}}(D)=\tfrac12\log_2(1+D)>0$ on
$(0,\tfrac12]$, and equal to the full marginalized value on
$[\tfrac12,1)$ (where $L_{\mathrm{vec}}=0$; the absolute gap is decreasing
there, but it is all of $L_{\mathrm{marg}}$).
\end{enumerate}
\end{proposition}

\begin{proof}
(i) is the reduction above with Paper V's scalar-corner formula at
$\sigma^2=1$, $\rho^2=\tfrac12$.

(ii) \emph{Converse.} For any admissible $\hY$ (Markov $\hY-X-S$),
$I(X;\hY\mid S)\ge I(Y;\hY\mid S)$, and the induced conditional channel
$p(\hat y\mid y,s)$ is feasible for Gray's conditional problem at the same
distortion, so $I(Y;\hY\mid S)\ge R_{Y\mid S}(D)=\tfrac12\log_2^{+}
\bigl(\operatorname{Var}(Y\mid S)/D\bigr)=\tfrac12\log_2^{+}(1/(2D))$.
\emph{Achievability.} For $D\le\tfrac12$ take
$\hY=(1-D)X_1+DX_2+N$: distortion $D^2+D^2+D(1-2D)=D$;
$\operatorname{Var}(\hY)=1-D$, $\Cov(\hY,S)=1$,
$\operatorname{Var}(\hY\mid S)=1-D-\tfrac12=\tfrac12-D$, so
$I(X;\hY\mid S)=\tfrac12\log_2\frac{(1-2D)/2}{D(1-2D)}
=\tfrac12\log_2\frac1{2D}$. For $D\ge\tfrac12$, $\hY=S/2$ is a
deterministic function of $X$, has distortion
$\E\bigl((X_1-X_2)/2\bigr)^2=\tfrac12\le D$, and is constant given $S$, so
$I(X;\hY\mid S)=0$.

(iii) Subtract. Numerical falsification (pre-assertion, logged): the
closed forms reproduce under direct minimization over all linear channels
(40-start SLSQP, five $D$ values, agreement $2\times10^{-4}$), and a
discretized conditional Blahut--Arimoto optimizer over \emph{unrestricted}
channels matches $L_{\mathrm{vec}}$ within grid error
($0.372$ vs $0.368$ at $D=0.3$; $0.005$ vs $0$ at $D=0.5$) while the
$Y$-only optimizer matches the corner ($0.560$ vs $0.558$; $0.293$ vs
$0.293$).
\end{proof}

\begin{remark}[The mechanism: writing in ink the eraser can read]
\label{rem:mechanism}
When $S$ is $X$-measurable, the encoder implicitly \emph{knows} $S$, and
the common-reconstruction Markov constraint costs nothing beyond Gray's
conditional problem: the optimal channel deliberately correlates the
record with $S$ --- at $D\ge\tfrac12$ the record \emph{is} a function of
the reset context, erasure-free by construction. Steinberg's scalar
strictness (CR $>$ conditional RD at $D>0$) is thus not intrinsic to the
CR constraint; it is the price of an encoder that cannot see $S$'s
non-read directions. General jointly Gaussian $S=\gamma^{\mathsf T}X+U$
($\operatorname{Var}U=\tau^2>0$) interpolates: the encoder commands the
$X$-part $\gamma^{\mathsf T}X$ but not $U$, suggesting the refined
question below. Operationally this is a new prediction for any GO-11
face: optimal joint records should tilt \emph{toward} the reset context's
$X$-measurable directions --- the encoder-side complement of the
eraser-side allocation tilt measured in GO-P-2026-058/059.
\end{remark}

\begin{question}[Refined form of Question~\ref{q:single} --- settled in
Section~\ref{sec:refined}]\label{q:refined}
For $S=\gamma^{\mathsf T}X+U$, is the vector CR function exactly a
Kaspi-type encoder-side-information object driven by the
encoder-accessible statistic $\gamma^{\mathsf T}X$ --- reducing to Gray at
$\tau^2=0$ (Proposition~\ref{prop:marg}) and to Steinberg's scalar formula
when $\gamma\parallel w$? \emph{Answer
(Theorem~\ref{thm:pair}--Corollary~\ref{cor:kaspi}): structurally yes ---
only the pair $(w^{\mathsf T}X,\gamma^{\mathsf T}X)$ matters, in any
dimension; formula-wise no --- the value is a new closed-form interpolating
family \eqref{eq:quadratic}, provably not a functional of the $(Y,S)$
margin.}
\end{question}

\section{The refined question, settled: pair sufficiency and the cubic
family}\label{sec:refined}

Normalize $\operatorname{Var}(w^{\mathsf T}X)=1$ (rescaling $D$), write
$Y=w^{\mathsf T}X$, $V=\gamma^{\mathsf T}X/\sigma_{\gamma}$ with
$\rho:=\E[YV]$, $|\rho|<1$, and rescale $S$ (invariantly for all mutual
informations) to $S=V+U$, $U\sim\mathcal N(0,\tau^2)$, $\tau^2>0$, with $X$, $U$, and all channel
randomness (including any resampling randomness below) \emph{mutually}
independent --- the joint form is what the $S$-kernel argument uses.

\begin{theorem}[Pair sufficiency]\label{thm:pair}
In the single-consumer problem
$L(D)=\min\{I(X;\hY\mid S):\hY-X-S,\ \E(Y-\hY)^2\le D\}$, it is without
loss of optimality to let the channel depend on $X$ only through
$T=(Y,V)$, and then $I(X;\hY\mid S)=I(T;\hY\mid S)$. Consequently $L(D)$
depends on $(\Sigma_x,w,\gamma,\tau^2)$, in any ambient dimension $d$,
only through $(\rho,\tau^2)$.
\end{theorem}
\begin{proof}
Given an admissible $p(\hat y\mid x)$, draw $\hY'$ from the conditional
law $p(\hat y\mid T=t)$ using fresh randomness. (a) $(T,\hY')\overset{d}=
(T,\hY)$, so the distortion (a functional of $(Y,\hY)$) is unchanged.
(b) $S=V+U$ with $U$ independent of $(X,\hY)$ and of $(X,\hY')$, so in
both joints $p(s\mid t,\hat y)=p_U(s-v)$: hence
$(T,\hY',S)\overset{d}=(T,\hY,S)$. (c) Given $T$, $\hY'$ is independent
of $(X,S)$, so $I(X;\hY'\mid S)=I(T;\hY'\mid S)+I(X;\hY'\mid T,S)
=I(T;\hY\mid S)\le I(X;\hY\mid S)$, the last step because $T$ is a
function of $X$. (d) $\hY'-X-S$ holds since $\hY'$ is a function of
$(X,\text{fresh noise})$ with the noise independent of $(X,U)$. The same
computation applied to a $T$-channel directly gives
$I(X;\hY\mid S)=I(T;\hY\mid S)$.
\end{proof}

\begin{theorem}[The exact single-consumer CR function, rank-one read]
\label{thm:cubic}
Write $s=1+\tau^2$. For $0<D<1$,
\begin{equation}
L(D)=\tfrac12\log_2 g^\star,\qquad
g^\star=\frac{(D+s-\rho^2)+\sqrt{(D+s-\rho^2)^2-4Ds(1-\rho^2)}}{2Ds},
\label{eq:quadratic}
\end{equation}
the larger root of $P(g)=Dsg^2-(D+s-\rho^2)g+(1-\rho^2)$; since
$P(1)=(D-1)\tau^2<0$, exactly one root exceeds $1$, so $g^\star$ is
well defined and unique. The optimum is attained by the jointly Gaussian
channel $\hY=aY+bV+N$, $a=\dfrac{g^\star-1}{g^\star}$,
$b=\dfrac{(g^\star-1)\rho}{g^\star k}$, $k=g^\star(1{+}\tau^2)-1$,
$N\sim\mathcal N\bigl(0,\;Q/(g^\star-1)\bigr)$,
$Q=a^2+b^2+2ab\rho-(a\rho+b)^2/(1{+}\tau^2)$. Anchors, algebraically:
$P$ factors as $(Dg-1)(sg-1)$ at $\rho=0$ (classical, $g^\star=1/D$); as
$(g-1)(Dg-(1-\rho^2))$ at $\tau^2\to0$ (Gray, matching
Proposition~\ref{prop:marg}, including the achieving channel); and gives
$g^\star=(D+\tau^2)/(D(1{+}\tau^2))$ at $\rho^2\to1$ (Steinberg's corner
at $\rho_{YS}^2=1/(1{+}\tau^2)$).
\end{theorem}
\begin{proof}[Proof (complete; converse closed by the verification pass)]
\emph{Achievability.} For the stated channel,
$I(T;\hY\mid S)=\tfrac12\log_2\bigl(\operatorname{Var}(\hY\mid S)/
\operatorname{Var}N\bigr)$ (jointly Gaussian; $S\perp N$ given $T$), and
$P(g^\star)=0$ is the statement that the distortion equals $D$ while
$\operatorname{Var}(\hY\mid S)/\operatorname{Var}N=g^\star$.

\emph{Converse.} By Theorem~\ref{thm:pair} restrict to $T$-channels. For
any admissible joint, with $c=\E[Y\hY]$, $q=\E[V\hY]$,
$v=\operatorname{Var}\hY$ (WLOG $\E\hY=0$): mutual independence of
$(X,U,\text{channel noise})$ forces the moment identity
$\E[S\hY]=\E[V\hY]=q$ (Gaussian Markovity at the second-moment level);
the distortion constraint reads $1-2c+v\le D$; and
$L\ge B(c,q,v):=\tfrac12\log_2
\bigl(\det\Sigma_{T\mid S}/\det\Sigma_e(c,q,v)\bigr)$, with $\Sigma_e$
the error covariance of the best \emph{linear} estimate of $T$ from
$(\hY,S)$ --- valid for arbitrary reproduction alphabets by Gaussian
entropy maximization. The minimization of $B$ closes exactly: in the
variables $(a,b)=\Sigma_T^{-1}(c,q)^{\mathsf T}$,
$n=v-(a^2+b^2+2ab\rho)$, the determinant identity
\[
\frac{\det\Sigma_{T\mid S}}{\det\Sigma_e}
=\frac{\operatorname{Var}(a Y+bV\mid S)+n}{n}
\]
holds (the $S$-column of the moment matrix equals the $V$-column plus
$\tau^2 e_S$), so the moment program \emph{is} the linear-channel program;
$B\to\infty$ on the PSD boundary $n\to0$ and at the distortion boundary,
the first-order system is linear in $(a,b)$ at fixed $g$ with the unique
solution above, and $P(1)<0$ pins the unique stationary $g^\star>1$.
Hence $\min B=\tfrac12\log_2 g^\star$, matched by achievability.
[Derivation completed analytically by the fresh-context verification
pass; record and instance-level numerics (agreement to $3\times10^{-9}$
bits at eight fresh instances; 25{,}500-point root-uniqueness sweep) in
the NOVELTY addendum.]
\end{proof}

\begin{remark}[Scope --- what is and is not settled]
Theorem~\ref{thm:cubic} answers Question~\ref{q:single}'s \emph{work-only
endpoint} for rank-one reads in full generality of
$(d,\Sigma_x,\gamma,\tau^2)$: by Theorem~\ref{thm:pair} the ambient
problem is exhausted by $(\rho,\tau^2)$, and Gaussian channels suffice
\emph{at that endpoint}. It does \emph{not} settle
Conjectures~\ref{conj:gauss}--\ref{conj:frontier}: the full $(R,L)$
frontier for $m=1$ is a genuinely nontrivial curve --- the $L$-optimal
channel pays strict excess rate over $R_{\min}=\tfrac12\log_2(1/D)$
(e.g.\ $0.040$ bits at $(\rho^2,\tau^2,D)=(0.75,0.5,0.3)$; verification
pass), so \emph{no single channel attains both corners}. This is itself a
finding: unlike Paper V's scalar corner, the rank-one-read problem with
non-aligned side information has a real rate--work tradeoff already at
$m=1$. The moment method visibly extends to the weighted objective
$\alpha R+(1-\alpha)L$ (both max-entropy bounds hold with the same three
moments) --- tracing that frontier is the named next step, before the
$m\ge2$ case, for which the pair-sufficiency argument generalizes
verbatim (``$(P$-read components, $\gamma^{\mathsf T}X)$ suffice''),
collapsing two rank-one consumers to at most three coordinates.
\end{remark}

\subsection*{The $m=1$ frontier}

\begin{theorem}[The exact rate--work region, $m=1$, rank-one read]
\label{thm:frontier}
For $\rho^2\in(0,1)$, $\tau^2>0$, $0<D<1$, the region
$\{(R,L)\}$ of Section~1 restricted to one consumer is the closed convex
set whose Pareto frontier is traced by $\alpha\in[0,1]$ as follows. Let
$(\gamma_0,\gamma_1)\in(0,1]^2$ solve, together with
$(a,b,n)$, the stationarity system
\[
a=1-\alpha\gamma_0-(1-\alpha)\gamma_1,\qquad
m=\frac{a\rho}{s-(1-\alpha)\gamma_1},\qquad
b=(1-\alpha)\gamma_1 m,
\]
\[
n=D-h(a,b),\qquad
\gamma_0=\frac{n}{Q_0+n},\qquad
\gamma_1=\frac{n}{Q_1+n},
\]
where $h(a,b)=(1-a)^2-2(1-a)b\rho+b^2$,
$Q_0=a^2+b^2+2ab\rho$, $Q_1=Q_0-(a\rho+b)^2/s$, and $m:=(a\rho+b)/s$
(solution attained by the weighted optimum; at $\alpha\in\{0,1\}$ the
minimizing moments are provably unique --- $P(1)<0$ at $\alpha=0$; the
elimination $b(1-\rho^2)=0$ then $(a,n)=(1{-}D,D(1{-}D))$ at $\alpha=1$ ---
and unique on every tested interior $\alpha$; completeness of the trace
holds wherever the weighted minimizer is unique). The frontier point is
\[
\bigl(R(\alpha),L(\alpha)\bigr)
=\Bigl(\tfrac12\log_2\tfrac1{\gamma_0},\ \tfrac12\log_2\tfrac1{\gamma_1}\Bigr),
\]
achieved by $\hY=aY+bV+N$, $N\sim\mathcal N(0,n)$. Endpoints:
$\alpha=1$ recovers the classical reverse channel
($b=0$, $a=1-D$, $R=\tfrac12\log_2\tfrac1D$); $\alpha=0$ recovers
Theorem~\ref{thm:cubic}'s channel and value.
\end{theorem}
\begin{proof}[Proof (converse mechanism as in Theorem~\ref{thm:cubic})]
Both coordinates are convex in the channel (the conditional coordinate by
the per-$s$ decomposition, VI-8), so the scalarization
$\alpha R+(1-\alpha)L$ traces the full Pareto boundary. For any admissible
channel with moments $(c,q,v)$, both max-entropy bounds hold on the same
three moments:
$R\ge\tfrac12\log_2(\det\Sigma_T/\det\Sigma_{e0})$ (estimation from $\hY$
alone) and $L\ge\tfrac12\log_2(\det\Sigma_{T\mid S}/\det\Sigma_{e1})$
(from $(\hY,S)$, with the Markov moment identity $\E[S\hY]=q$). The two
determinant identities
\[
\frac{\det\Sigma_T}{\det\Sigma_{e0}}=\frac{Q_0+n}{n},
\qquad
\frac{\det\Sigma_{T\mid S}}{\det\Sigma_{e1}}=\frac{Q_1+n}{n}
\]
(the first by the same Schur computation as the second) make the weighted
moment program identical to the weighted linear-channel program; it is
coercive on its boundaries, and its first-order system --- using
$Q_{0,a}-h_a=2$, $Q_{0,b}-h_b=2\rho$, $Q_{1,a}-h_a=2(1-\rho m)$,
$Q_{1,b}-h_b=2(\rho-m)$ --- reduces linearly to the displayed relations.
Achievability is the stated Gaussian channel, which meets both bounds with
equality. [Numerics logged pre-assertion, seed 20260806: the system holds
at the multi-start optimum and the moment program agrees to $10^{-6}$
bits at 20 $(\rho^2,\tau^2,D,\alpha)$ combinations; the $\alpha$-sweep is
monotone between the corners.]
\end{proof}

\begin{corollary}[Misalignment always opens a tradeoff]\label{cor:sep}
For $\rho^2\in(0,1)$ and $\tau^2>0$ the frontier is nondegenerate. The
$L$-optimal moments are unique with $b\ne0$ (Theorem~\ref{thm:cubic};
$b=(g^\star{-}1)\rho/(g^\star k)\ne0$), and the $R$-optimal moments are
unique with $b=0$ (at $\alpha=1$ the elimination gives $b(1-\rho^2)=0$,
and then $(a,n)=(1{-}D,\,D(1{-}D))$ is forced). Hence any channel
attaining $L_{\min}$ carries the $b\ne0$ moments, so its rate is at least
$B_0$ evaluated there, which strictly exceeds $R_{\min}$ ($B_0$'s argmin
being unique with $b=0$) --- and symmetrically. Both endpoint excesses
are strictly positive (at $(\rho^2,\tau^2,D)=(0.75,0.5,0.3)$:
$R(0)-R_{\min}=0.0400$, $L(1)-L_{\min}=0.0349$ bits). The single-corner
collapses within the parameter space are exactly the alignment
degeneracies: $\rho^2\to1$ (Paper V's scalar corner) and $\rho=0$ (the
frontier collapses to the single corner $L_{\min}=R_{\min}$; $\tau^2\to
\infty$ is the same collapse in the limit). Closed-form endpoint values:
$L(1)=\tfrac12\log_2\bigl[((1-D)(1-\rho^2/s)+D)/D\bigr]$ and $R(0)$ from
Theorem~\ref{thm:cubic}'s channel. The same strictness holds in the
$\tau^2=0$ instance computations of Proposition~\ref{prop:marg} for
$D<\tfrac12$
(e.g.\ $R(0)=\tfrac12\log_2[(1-D)/(D(1-2D))]>\tfrac12\log_2(1/D)$), so
even a fully $X$-measurable context trades rate against work once
misaligned.
\end{corollary}

This settles Conjecture~\ref{conj:gauss} affirmatively for $m=1$,
rank-one reads (Gaussian sufficiency for the whole frontier), and
replaces Conjecture~\ref{conj:frontier}'s $m=1$ shadow with the explicit
two-water-level system; the $m\ge2$ statements remain open.

\begin{corollary}[Answer to the refined question]\label{cor:kaspi}
(i) \emph{Structurally yes}: the value is driven entirely by the
encoder-accessible statistic $\gamma^{\mathsf T}X$ (pair sufficiency),
reducing to Gray at $\tau^2=0$ and to Steinberg at $\gamma\parallel w$.
(ii) \emph{Formula-wise no}: $L$ is \emph{not} a functional of the joint
law of $(Y,S)$ --- the instances $(\rho^2,\tau^2)=(\tfrac12,0)$ and
$(\tfrac34,\tfrac12)$ have identical $(Y,S)$ margins
($\rho_{YS}^2=\tfrac12$) yet $L=1.1610$ vs $1.2105$ bits at $D=0.1$
($0.3685$ vs $0.5228$ at $D=0.3$), both strictly below the
Steinberg value of the margin ($1.2297$; $0.5577$). In particular the
vector CR function is not Steinberg's formula composed with any modified
pair, nor any known scalar object: it is the new interpolating family
\eqref{eq:quadratic}, monotone (observed) in the encoder-access split
$\tau^2$ at fixed margin.
\end{corollary}

\section{Delineation (sweeps of 2026-08-04, record in
\texttt{go11-conditional-region-NOVELTY.md})}\label{sec:delineation}

\textbf{In-repo (Paper V).} The one-constraint pair-region is Paper V's,
completely: main region theorem (general alphabets), Pareto-channel fixed
point (= our scalarization at $\beta'=1-\alpha$), scalar corner, and
mode-aligned reset water-filling. Paper V's multi-consumer theorem gives
only the structural union for $m\ge2$; its ``general Gaussian coupling''
limitation names Question~\ref{q:single}'s territory as open.

\textbf{IB / task-oriented compression.} The unconstrained Gaussian
tradeoff of \eqref{eq:dcib} is the solved Gaussian information bottleneck
\cite{gib2005}. The \emph{scalar single-distortion instance of our
frontier is prior art}: Wang et al.'s rate--distortion--classification
function \cite{wang2024rdc} minimizes $I(X;\hX)$ under an MSE cap and an
$I(\hX;S)$ floor (their $H(S\mid\hX)\le C$), with our Markov chain, solved
in closed scalar/bivariate-Gaussian form --- including the
one-constraint-active-at-a-time structure that matches the scalar corner.
The strongest machinery match is Liu et al.'s semantic rate--distortion
function \cite{liu2022semantic}: two \emph{simultaneous} quadratic
constraints on a Gaussian source, solved as an error-covariance convex
program with reverse water-filling --- but its second coordinate is an MSE
on the state, not an information quantity. The sign-flipped
(minimize-leakage) family has scalar Gaussian closed forms since
Rebollo-Monedero et al.\ (2010) and Sankar et al.\ \cite{sankar2013}, and a
special-case vector water-filling tilting distortion toward the modes that
leak least \cite{tripathy2019} --- the mirror of
Conjecture~\ref{conj:frontier}'s allocation.

\textbf{Secrecy / equivocation (the same family, same orientation).}
Maximizing source equivocation at an $S$-holding party is minimizing our
$L$ up to $H(X\mid S)$, so the Yamamoto lineage \cite{yamamoto1983} is not
a cousin but the family itself. Exact Gaussian regions exist and are all
scalar with one distortion constraint: Villard--Piantanida
\cite{villard2013} (exact only when the eavesdropper has no side
information; the non-degraded scalar Gaussian converse \emph{explicitly
left open}); Schieler--Cuff (Gaussian solved only without adversary side
information); and, sharpest, G\"unl\"u--Schaefer--Boche--Poor
\cite{gunlu2023}, whose privacy-leakage coordinate $I(X;W\mid S)$ \emph{is}
our conditional coordinate, solved in closed scalar-Gaussian form via the
conditional EPI under a degradedness ordering --- containing the scalar
single-consumer corner. The only vector-Gaussian equivocation result found
(Ekrem--Ulukus \cite{ekrem2013}) is tight only with the rate constraint
removed. The only exact two-distortion-plus-equivocation region found
(Tandon--Sankar--Poor) is discrete-alphabet, and it privatizes $S$ itself.

\textbf{Common reconstruction (Question~\ref{q:single}'s endpoint).}
Steinberg \cite{steinberg2009} and Lapidoth--Mal\"ar--Wigger
\cite{lmw2014} solve the \emph{scalar} Gaussian CR function (the latter
for arbitrary scalar jointly Gaussian $S$ via their $\xi$-reduction); the
entire CR follow-up line (Heegard--Berger-with-CR, cascade, successive
refinement, interactive, action-dependent) contains no vector Gaussian
case. The general-coupling vector machinery that exists
(Stylianou--Gkagkos--Charalambous's indirect/WZ RDFs with side information)
derives its water-filling precisely from the estimator's \emph{use} of $S$
--- the step the CR Markov constraint forbids.

\textbf{Summary of the open set.} Under the recorded queries: no prior
treats the pair $(I(X;\hX),\,I(X;\hX\mid S))$ as a two-coordinate region
beyond Paper V's one-constraint solution; no exact Gaussian region combines
two quadratic distortion constraints with a conditional-information
coordinate; no non-degraded vector-Gaussian converse exists for any member
of the family; no vector-Gaussian CR function is in print. Residual risks:
IEEE-only venues without arXiv copies; the fast-moving RDC line
\cite{wang2024rdc} producing a vector-Gaussian paper; the
maximal-correlation area received the weakest coverage.

\section{The $m=2$ region, settled as a matrix program}\label{sec:m2}

Two rank-one consumers ($Y_A=u^{\mathsf T}X$, $Y_B=v^{\mathsf T}X$,
normalized unit variances, read correlation $\rho_{AB}$), scalar jointly
Gaussian reset context $S=V+U$ as before, correlations
$(\rho_{AV},\rho_{BV})$; WLOG scalar reproductions
$(\hY_A,\hY_B)$ (per-coordinate data processing, as in
Section~\ref{sec:marg}).

\begin{theorem}[$(m{+}1)$-sufficiency]\label{thm:triple}
For $m$ rank-one consumers, it is without loss of optimality to let the
joint channel depend on $X$ only through
$T=(Y_1,\ldots,Y_m,V)$, and then both coordinates equal their
$T$-versions. The proof of Theorem~\ref{thm:pair} applies verbatim: the
resampled channel preserves the joint law of $(T,\hat{\mathbf Y},S)$ because
$S=V+U$ with $U$ jointly independent of $(X,\text{all channel
randomness})$, and distortions are functionals of $(Y_i,\hY_i)$ pairs.
For $m=2$ the ambient problem, in any dimension $d$, is exhausted by
$(\rho_{AB},\rho_{AV},\rho_{BV},\tau^2)$.
\end{theorem}

\begin{theorem}[The exact $m=2$ region]\label{thm:m2}
Assume $\Sigma_T\succ0$ (degenerate instances reduce: if $V$ is
$Y_A$-measurable, drop $V$ and run the identities on the reduced pair
with $\Sigma_{T'\mid S}$ in place of $\Sigma_{T\mid S}$). Let $\Sigma_T$
be the $3\times3$ correlation matrix of $T$ and
$\Sigma_{T\mid S}=\Sigma_T-\sigma_{TS}\sigma_{TS}^{\mathsf T}/(1+\tau^2)$,
$\sigma_{TS}=\Sigma_T e_V$. Then
\[
\mathcal{RW}(D_A,D_B\mid S)=\operatorname{cl}\!\!
\bigcup_{\substack{A\in\mathbb R^{2\times3},\ \Sigma_N\succeq0\\
(e_i-A_{i\cdot})^{\mathsf T}\Sigma_T(e_i-A_{i\cdot})+(\Sigma_N)_{ii}\le D_i}}
\Bigl\{(R,L):\,
R\ge\tfrac12\log_2\tfrac{\det(A\Sigma_TA^{\mathsf T}+\Sigma_N)}{\det\Sigma_N},\
L\ge\tfrac12\log_2\tfrac{\det(A\Sigma_{T\mid S}A^{\mathsf T}+\Sigma_N)}{\det\Sigma_N}
\Bigr\}:
\]
jointly Gaussian channels $\hat{\mathbf Y}=AT+N$, $N\sim\mathcal N(0,\Sigma_N)$,
exhaust the region, whose frontier is a nine-parameter matrix program ---
the work-coordinate analog of the simultaneous-service max-det program of
\cite{bondtwoobserver}.
\end{theorem}
\begin{proof}
\emph{Converse.} By Theorem~\ref{thm:triple} restrict to $T$-channels.
For any admissible channel let $C=\Cov(T,\hat{\mathbf Y})$ ($3\times2$),
$\Sigma_{\hat Y}=\Cov(\hat{\mathbf Y})$, and set $A:=C^{\mathsf T}\Sigma_T^{-1}$,
$\Sigma_N:=\Sigma_{\hat Y}-A\Sigma_TA^{\mathsf T}$ (the Schur complement;
joint PSD $\Leftrightarrow\Sigma_N\succeq0$). The Markov chain and joint
independence of $U$ force $\Cov(S,\hat{\mathbf Y})=\Cov(V,\hat{\mathbf Y})$
(the $V$-row of $C$). Max-entropy bounds on these shared moments give
$R\ge\tfrac12\log_2(\det\Sigma_T/\det\Sigma_{e0})$ and
$L\ge\tfrac12\log_2(\det\Sigma_{T\mid S}/\det\Sigma_{e1})$, and the two
determinant identities generalize by the same Schur computation:
$\det\Sigma_T/\det\Sigma_{e0}
=\det(A\Sigma_TA^{\mathsf T}+\Sigma_N)/\det\Sigma_N$ and likewise
conditionally (the $S$-column of the moment matrix is the $V$-column plus
$\tau^2e_S$). The distortion constraints are exactly the displayed moment
constraints. \emph{Achievability.} The Gaussian channel with parameters
$(A,\Sigma_N)$ realizes those moments and attains both bounds with
equality. \emph{Numerics (pre-assertion, seed 20260807):} the $\alpha=1$
projection reproduces the on-regime Xiao--Luo value
$\tfrac12\log_2\bigl((1-\rho_{AB}^2)/(D_AD_B)\bigr)$ to $10^{-5}$;
the tax-note floors hold at every computed point; a coarse unrestricted
conditional-BA with two-dimensional reproduction never materially beats
the program (grid bias disclosed in the NOVELTY record).
\end{proof}

\begin{remark}[No closed form: Conjecture~\ref{conj:frontier} withdrawn
as stated]\label{rem:noclosed}
The $m=1$ collapse to a scalar quadratic does not persist: the frontier
program couples three mutually non-commuting quadratic forms
($\Sigma_T$, $\Sigma_{T\mid S}$, and the constraint directions), and no
water-filling closed form emerged --- precisely the outcome
Conjecture~\ref{conj:frontier}'s caution anticipated. We therefore
\emph{withdraw Conjecture~\ref{conj:frontier} as stated} and replace it
with Theorem~\ref{thm:m2}'s program, which is the honest $m=2$ endpoint
of this line: exact, finite-dimensional, solver-computable. Alignment
degeneracies still collapse it (next corollary), and
Conjecture~\ref{conj:gauss} is settled affirmatively for $m=2$ rank-one
by Theorem~\ref{thm:m2} itself.
\end{remark}

\begin{corollary}[The GO-10 worked instance, exactly]\label{cor:go10}
For orthogonal reads ($\rho_{AB}=0$) with $S$ aligned to consumer $A$
($V=Y_A$), the region is the single-corner quadrant at
$\bigl(R_A+R_B,\ L_A+L_B\bigr)$ with $R_A=R_B=\tfrac12\log_2(1/D)$,
$L_B=R_B$, and $L_A$ Steinberg's corner at $\rho_s^2=1/(1+\tau^2)$;
consequently the complementarity-tax gap is exactly
\[
\mathrm{CT}_R-\mathrm{CT}_W=\tfrac12\log_2\frac1{s^2+(1-s^2)D},
\qquad s^2=\frac{\tau^2}{1+\tau^2},
\]
the formula of \cite[Sec.~5]{bondtaxnote}, now derived from the exact
region rather than the per-coordinate reduction (numerically: all four
quantities to $4$ decimals at $D\in\{0.25,0.1\}$, $\tau^2=0.25$, and at
$\tau^2=0.6$, $D=0.15$, with the $\alpha=0$ and $\alpha=1$ optima
coinciding --- corner degeneracy --- as alignment requires).
\end{corollary}
\begin{proof}
This instance is $\Sigma_T$-degenerate ($V=Y_A$); apply
Theorem~\ref{thm:m2}'s reduced form on $T'=(Y_A,Y_B)$, $\Sigma_{T'}=I_2$,
$\Sigma_{T'\mid S}=\mathrm{diag}(s^2,1)$. \emph{Achievability}:
independent per-consumer channels. \emph{Super-additivity converse}:
$(Y_A,S)\perp Y_B$ (since $S=Y_A+U$, $Y_B\perp Y_A$, $U$ jointly
independent), so for any admissible channel
$I(T';\hat{\mathbf Y}\mid S)=I(Y_A;\hat{\mathbf Y}\mid S)
+I(Y_B;\hat{\mathbf Y}\mid Y_A,S)\ge
I(Y_A;\hat{\mathbf Y}\mid S)+I(Y_B;\hat{\mathbf Y}\mid S)$, and each
induced marginal channel \emph{is} Markov
($\hat{\mathbf Y}=f(Y_A,W)$ with $W=(Y_B,\text{noise})\perp(Y_A,S)$, and
symmetrically), so Steinberg's scalar converse prices the $A$-term and
the classical converse the $B$-term; the same decomposition without $S$
handles $R$. The Markov step holds \emph{only} by the instance's double
orthogonality ($\rho_{AB}=\rho_{BV}=0$) --- it is exactly the coupling
that Proposition~\ref{prop:marg} shows can fail in general.
\end{proof}

\begin{theorem}[Work-floor exactness: Conjecture~\ref{conj:tight} settled,
with a correction]\label{thm:c3}
Write $\Sigma_Y$ for the read-pair covariance, $\beta=\Cov(Y,V)\in\mathbb
R^2$, $\Sigma_{Y\mid V}=\Sigma_Y-\beta\beta^{\mathsf T}$,
$\Sigma_{Y\mid S}=\Sigma_Y-\beta\beta^{\mathsf T}/(1{+}\tau^2)$,
$\Delta=\mathrm{diag}(D_A,D_B)\succ0$. The work floor
$\tfrac12\log_2\bigl(\det\Sigma_{Y\mid S}/\det\Delta\bigr)$ is attained by
the $m{=}2$ program iff
\[
\text{(a) } \tau^2=0 \text{ and } \Delta\preceq\Sigma_{Y\mid V},
\qquad\text{or}\qquad
\text{(b) } \beta=0 \text{ and } \Delta\preceq\Sigma_Y .
\]
In every other case the floor is strict --- both on the misaligned branch
($\tau^2>0$ and $\beta\ne0$, at every positive $(D_A,D_B)$) and outside
the semidefinite conditions of (a)/(b) --- with, on the misaligned branch
for $\Delta\prec\Sigma_{Y\mid V}$, the deficit sandwich
\[
0\;<\;L_{\min}-\mathrm{floor}\;\le\;
\tfrac12\log_2\frac{\det\bigl(I-\Sigma_{Y\mid S}^{-1}\Delta\bigr)}
{\det\bigl(I-\Sigma_{Y\mid V}^{-1}\Delta\bigr)}
\;=\;\tfrac12\,\tr\bigl[(\Sigma_{Y\mid V}^{-1}-\Sigma_{Y\mid S}^{-1})
\Delta\bigr]\log_2 e+O(\|\Delta\|^2)\;\longrightarrow\;0
\ \ (\Delta\to0).
\]
\emph{The correction:} Conjecture~\ref{conj:tight} guessed the condition
$\Delta\preceq\Sigma_{Y\mid S}$; the attainable branch in fact requires
the \emph{encoder-accessible} conditional $\Sigma_{Y\mid V}$ --- the same
$S$-vs-$V$ substitution that Proposition~\ref{prop:marg} and
Theorem~\ref{thm:cubic} forced everywhere else in this program.
\end{theorem}
\begin{proof}
\emph{Necessity.} Suppose the floor is attained. Equality in the floor's
chain forces, at the moment level (max-entropy tightness makes the joint
WLOG Gaussian): $E_S:=\E[\Cov(Y\mid\hat{\mathbf Y},S)]=\Delta$ with
per-consumer equality, so $Z:=Y-\hat{\mathbf Y}$ has $\Cov Z=\Delta$,
$\Cov(Z,\hat{\mathbf Y})=0$, $\Cov(Z,S)=0$; the latter with the Markov
moment identity gives $\Cov(Z,V)=0$, hence
$\Cov(\hat{\mathbf Y},V)=\beta$. Since $E_S\preceq\Sigma_{Y\mid S}$ always,
$\Delta\preceq\Sigma_{Y\mid S}$ is necessary; at $\tau^2=0$ this is
$\Delta\preceq\Sigma_{Y\mid V}$. Tightness of
$I(T;\hat{\mathbf Y}\mid S)=I(Y;\hat{\mathbf Y}\mid S)$ additionally
requires $\Cov(V,\hat{\mathbf Y}\mid Y,S)=0$; with the pinned moments this
evaluates to
$[\Delta\ \ 0]\,M^{-1}\bigl[\beta;\ \Cov(S,V)\bigr]=0$, $M=\Cov(Y,S)$,
which (for $\Delta\succ0$) holds iff $\beta=0$ or
$\Cov(S,V)=\operatorname{Var}S$, i.e.\ $\tau^2=0$. \emph{Sufficiency (a).}
At $\tau^2=0$ take $Z\sim\mathcal N(0,\Delta)$ jointly Gaussian with
$\Cov(Z,Y)=\Delta$, $\Cov(Z,V)=0$ --- existent iff
$\Delta-\Delta\Sigma_{Y\mid V}^{-1}\Delta\succeq0
\Leftrightarrow\Delta\preceq\Sigma_{Y\mid V}$ (Schur) --- and set
$\hat{\mathbf Y}=Y-Z$: then $\Cov(\hat{\mathbf Y}\mid S)
=\Sigma_{Y\mid S}-\Delta$ and
$\Cov(\hat{\mathbf Y}\mid T,S)=\Delta-\Delta\Sigma_{Y\mid V}^{-1}\Delta$,
and at $\tau^2=0$ ($\Sigma_{Y\mid S}=\Sigma_{Y\mid V}=:\Sigma_c$)
\[
L=\tfrac12\log_2\frac{\det(\Sigma_c-\Delta)}
{\det\Delta\,\det(I-\Sigma_c^{-1}\Delta)}
=\tfrac12\log_2\frac{\det\Sigma_c}{\det\Delta}=\mathrm{floor}.
\]
\emph{Sufficiency (b).} At $\beta=0$, $\Sigma_{Y\mid S}=\Sigma_Y$ and a
plane-only channel has $\hat{\mathbf Y}\perp S$, so $L=I(Y;\hat{\mathbf
Y})$, attained at the floor iff $\Delta\preceq\Sigma_Y$ (the rate-side
exactness of \cite{bondtaxnote}); necessity of $\Delta\preceq\Sigma_Y$ is
the $E_S\preceq\Sigma_{Y\mid S}=\Sigma_Y$ bound. \emph{Sandwich.} The same
$Z$-construction at $\tau^2>0$ is admissible whenever
$\Delta\prec\Sigma_{Y\mid V}$ and yields
$L\le\tfrac12\log_2[\det(\Sigma_{Y\mid S}-\Delta)/\det(\Delta-
\Delta\Sigma_{Y\mid V}^{-1}\Delta)]=\mathrm{floor}
+\tfrac12\log_2[\det(I-\Sigma_{Y\mid S}^{-1}\Delta)/\det(I-
\Sigma_{Y\mid V}^{-1}\Delta)]$; the ratio is $\ge1$ because
$\Sigma_{Y\mid S}\succeq\Sigma_{Y\mid V}$ (congruence by $\Delta^{1/2}$
keeps both factors in the PSD order where $\det$ is monotone), and $\to1$
as $\Delta\to0$. Strictness ($>0$) is the necessity direction, using that
$L_{\min}$ is \emph{attained} (compact feasible $(A,\Sigma_N)$ set, lower
semicontinuity); at the non-strict boundary of
$\Delta\preceq\Sigma_{Y\mid V}$ in case (a), where the displayed ratio is
$0/0$, attainment follows by $\Delta_t\uparrow\Delta$ feasibility and
floor continuity. \emph{(m=1 corollary.)} For one consumer, on the
active-floor regime $D\le\operatorname{Var}(Y\mid V)$, the floor
$\tfrac12\log_2(\operatorname{Var}(Y\mid S)/D)$ is attained iff
$\rho\tau=0$: substituting $g_f=(s-\rho^2)/(Ds)$ into
Theorem~\ref{thm:cubic}'s quadratic gives $P(g_f)=-\rho^2\tau^2/s$, which
is $<0$ (hence $g^\star>g_f$ strictly) exactly when $\rho\tau\ne0$. (The
regime qualifier is the m=1 shadow of condition (a): for $\tau=0$,
$D>1-\rho^2$ the clamped floor and $L=0$ meet trivially while $g_f<1$
becomes the quadratic's \emph{smaller} root.)
\end{proof}
