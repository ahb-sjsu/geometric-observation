\title{GO-11 (problem statement): The Exact Two-Consumer\\
Rate--Work Region with Reset Side Information}
\author{Andrew H. Bond}
\date{Problem statement v0.1 --- August 2026}
\begin{document}\maketitle

\begin{abstract}
The complementarity-tax note proved \emph{floors} for the joint description
of two consumers whose reset may consult side information: the work
coordinate obeys $D_AD_B\ge\kappa_S\,2^{-2L}$, and the rate and work floors
differ by exactly $I(u^{\mathsf T}X,v^{\mathsf T}X;S)$. Those floors are
tight only as $D\to0$. This document poses the exact-region problem, records
the one organizing identity that shapes it --- per test channel, the
rate--work gap is \emph{exactly} $I(\hX;S)$, so the Pareto frontier is a
distortion-constrained information bottleneck with $S$ as relevance ---
states the conjectured Gaussian solution, and lays out the verification
obligations a future registration must meet. Nothing beyond Fact~1 is
claimed.
\end{abstract}

\section{The problem}
$X\sim\mathcal N(0,\Sigma_x)$ on $\mathbb R^d$, i.i.d.; consumers $A,B$ with
rank-one reads $u,v$ (general PSD $P_1,P_2$ as the extension); reset side
information $S$, jointly Gaussian with $X$, available only at erasure, with
the model Markov chain $(\hX_A,\hX_B)-X-S$ \cite{bondlandauerV}. Define, for
$0<D_A\le\sigma_u^2$, $0<D_B\le\sigma_v^2$,
\[
\mathcal{RW}(D_A,D_B\mid S)=\operatorname{cl}\!\bigcup_{p(\hat x_A,\hat x_B\mid x)}
\bigl\{(R,L):\ R\ge I(X;\hX_A,\hX_B),\ L\ge I(X;\hX_A,\hX_B\mid S)\bigr\},
\]
the union over joint test channels with $\E\,d_A\le D_A$, $\E\,d_B\le D_B$.

\textbf{Problem (GO-11).} Characterize $\mathcal{RW}(D_A,D_B\mid S)$
exactly --- its Pareto frontier, the optimal channel family, and the
condition under which the complementarity-tax floors
\cite{bondtaxnote} are attained.

\textbf{What is known.}
(i) The rate-only projection is the simultaneous-service max-det program
(\cite{bondtwoobserver}; after the read-plane reduction, the Xiao--Luo /
semidefinite-condition literature). (ii) The single-consumer scalar case is
Paper V's scalar Gaussian corner (single-corner quadrant, no tradeoff);
independent read modes give its reset water-filling frontier
\cite{bondlandauerV}. (iii) The two-consumer floors
$R\ge\tfrac12\log^+(\kappa/(D_AD_B))$,
$L\ge\tfrac12\log^+(\kappa_S/(D_AD_B))$, with floor difference exactly
$I(Y;S)$, $Y=B^{\mathsf T}X$; the rate floor is exact iff
$\mathrm{diag}(D_A,D_B)\preceq B^{\mathsf T}\Sigma_xB$, and the work floor
is generally strict at $D>0$ \cite{bondtaxnote}. (iv) Operationally, decode
thresholds realize $0.4$--$0.8\times$ of the floor-level gaps at finite $n$
on two source families (GO-P-2026-058/059) --- an exact region would say
what the remaining fraction is made of.

\section{The organizing identity}
\begin{fact}\label{fact:gap}
For any test channel with $(\hX_A,\hX_B)-X-S$ Markov, writing
$\hX=(\hX_A,\hX_B)$,
\[
I(X;\hX)-I(X;\hX\mid S)\;=\;I(\hX;S).
\]
\end{fact}
\begin{proof}
$I(\hX;S)=I(\hX;X,S)-I(\hX;X\mid S)=I(\hX;X)+I(\hX;S\mid X)-I(\hX;X\mid S)$,
and $I(\hX;S\mid X)=0$ by the Markov chain.
\end{proof}

So every channel contributes the corner $(R,L)=(I,\,I-I(\hX;S))$ with
$I=I(X;\hX)$: the region is generated by a \emph{single curve} of corners,
and its lower-left frontier is traced by
\begin{equation}
\min_{p(\hat x\mid x)}\ I(X;\hX)-\beta\, I(\hX;S)
\quad\text{s.t.}\quad \E d_A\le D_A,\ \E d_B\le D_B,
\qquad \beta\in[0,1],
\label{eq:dcib}
\end{equation}
a \emph{distortion-constrained information bottleneck} with the reset
context $S$ playing relevance: $\beta=0$ recovers the simultaneous-service
rate program; $\beta=1$ is the work-only endpoint. The unconstrained
Gaussian tradeoff of \eqref{eq:dcib} is the solved Gaussian information
bottleneck (eigen-structured linear encoders with a critical-$\beta$
phase structure) \cite{gib2005}; the open content of GO-11 is the
interaction of that structure with two simultaneous weighted-quadratic
distortion constraints.

\section{Conjectures}
\begin{conjecture}[Gaussian sufficiency]\label{conj:gauss}
The frontier of \eqref{eq:dcib} is attained by jointly Gaussian channels
$\hX = KX+W$; consequently the region is the closure of
$\{(R(\Sigma),\,R(\Sigma)-I_{\Sigma}(\hX;S))\}$ over error covariances
$0\preceq\Sigma\preceq\Sigma_x$ meeting the trace constraints, with both
coordinates explicit determinant functionals of $\Sigma$.
\end{conjecture}

\begin{conjecture}[Two-consumer reset water-filling]\label{conj:frontier}
In the $S$-whitened read frame, the frontier is traced by an
$\alpha$-parametrized allocation program generalizing Paper V's reset
water-filling: the $\alpha=1$ endpoint re-derives the max-det rate optimum,
the $\alpha=0$ endpoint tilts distortion toward the directions of the read
plane that $S$ resolves least, and for scalar-pair instances the program
reduces to a two-mode quadratic of the Paper V vector-frontier type.
\end{conjecture}

\begin{conjecture}[Exactness condition for the work floor]\label{conj:tight}
The work floor $\tfrac12\log(\kappa_S/(D_AD_B))$ is attained iff a
semidefinite condition of the form
$\mathrm{diag}(D_A,D_B)\preceq B^{\mathsf T}\Sigma_{X\mid S}B$ holds
\emph{and} the optimal channel decouples from $S$'s residual --- expected
to fail at every $D>0$ when $S$ is informative (matching the strictness
remark of \cite{bondtaxnote} and the scalar corner's $D>0$ gap), with
equality exactly in the $D\to0$ limit.
\end{conjecture}

\begin{question}[Single-consumer vector case]\label{q:single}
Even $m=1$ is open in general: for one rank-one read of a $d$-dimensional
source with \emph{arbitrary} jointly Gaussian $S$ (not aligned with the
read), what is $L_A(D\mid S)$? The scalar corner covers $S$ correlated with
the read projection only; the general case is the natural first theorem.
\end{question}

\begin{question}[Operational face]\label{q:oper}
Do decode thresholds track the exact frontier (not just the floors) --- in
particular, does the measured $0.4$--$0.8\times$ realized fraction of
GO-P-2026-058/059 decompose into (exact-region gap) $+$ (finite-$n$
instrument loss), with the first term predicted by
Conjecture~\ref{conj:frontier}?
\end{question}

\section{Line of attack and verification obligations}
\textbf{Attack.} (1) Prove Question~\ref{q:single} by the scalar corner's
moment/LMMSE method (the Markov identity
$\Cov(S,\hX)=\rho\,\Cov(X,\hX)$-style parametrization) lifted to the
$S$-whitened frame. (2) For Conjecture~\ref{conj:gauss}, the per-$s$
convexity decomposition of $I(X;\hX\mid S)$ (VI-8) plus the entropy-power
or eigen-alignment arguments of the Gaussian IB \cite{gib2005}. (3) For
Conjecture~\ref{conj:frontier}, KKT of the two-determinant program;
numerical falsification first (the 044/058 conditional Blahut--Arimoto
fixed point brute-forces small discretized instances against the
conjectured frontier \emph{before} any proof is attempted).
